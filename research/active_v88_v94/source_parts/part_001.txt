def metrics(returns: pd.Series, turnover: pd.Series | None = None, gross: pd.Series | None = None) -> dict:
    returns = returns.fillna(0.0).astype(float)
    if returns.empty:
        return {"total_return": 0.0, "cagr": 0.0, "max_drawdown": 0.0, "sharpe": 0.0, "annual_turnover": 0.0, "average_gross": 0.0, "max_gross": 0.0, "final_equity": 10000.0}
    equity = (1.0 + returns).cumprod() * 10000.0
    years = max(((equity.index[-1] - equity.index[0]).days + 1) / 365.0, 1 / 365.0)
    drawdown = equity / equity.cummax() - 1.0
    sd = float(returns.std(ddof=1))
    return {
        "total_return": float(equity.iloc[-1] / 10000.0 - 1.0),
        "cagr": float((equity.iloc[-1] / 10000.0) ** (1.0 / years) - 1.0),
        "max_drawdown": float(drawdown.min()),
        "sharpe": float(returns.mean() / sd * np.sqrt(365.0)) if sd > 0 else 0.0,
        "annual_turnover": float(turnover.fillna(0.0).sum() / years) if turnover is not None else 0.0,
        "average_gross": float(gross.fillna(0.0).mean()) if gross is not None else 0.0,
        "max_gross": float(gross.fillna(0.0).max()) if gross is not None else 0.0,
        "final_equity": float(equity.iloc[-1]),
    }


def yearly_returns(returns: pd.Series) -> pd.DataFrame:
    rows = []
    for year, group in returns.fillna(0.0).groupby(returns.index.year):
        rows.append({"year": int(year), "return": float((1.0 + group).prod() - 1.0)})
    return pd.DataFrame(rows)


def concentration(returns: pd.Series) -> dict:
    annual = yearly_returns(returns)
    positive_logs = [math.log1p(float(x)) for x in annual["return"] if x > 0]
    best_share = max(positive_logs) / sum(positive_logs) if positive_logs and sum(positive_logs) > 0 else 0.0
    loo = []
    for year in annual["year"]:
        sample = returns[returns.index.year != int(year)]
        loo.append({"excluded_year": int(year), "cagr": metrics(sample)["cagr"]})
    return {
        "best_positive_year_log_share": float(best_share),
        "worst_leave_one_year_out_cagr": float(min((x["cagr"] for x in loo), default=0.0)),
        "leave_one_year_out": loo,
    }


def rebalance_mask(index: pd.DatetimeIndex, every: int) -> np.ndarray:
    return np.arange(len(index)) % int(every) == 0


def cap_group(weights: pd.Series, columns: Iterable[str], cap: float) -> None:
    cols = [c for c in columns if c in weights.index]
    gross = float(weights.loc[cols].abs().sum())
    if gross > cap and gross > 0:
        weights.loc[cols] *= cap / gross
